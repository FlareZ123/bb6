"""Machine #746 source-aligned return reduction."""
from __future__ import annotations

def source_next(a: int, b: int) -> tuple[int, int] | None:
    if b == 0: return 2, 2*a+1
    if b == 1: return None
    d=b-2
    if d>=2*a: return 2*a+2,d-2*a
    t=6*a+d+(5 if d%2 else 8)
    return t//4,t%4

def v2(x: int) -> int:
    assert x>0
    return (x & -x).bit_length()-1

def drain(a: int) -> tuple[int, int]:
    k=v2(a+3)
    return (3**(k+1)*((a+3)>>k)-5)//2,k+1

def return_map(h: int) -> tuple[int | None, int, int, int]:
    assert h>=0
    n,p=0,1
    while h+n+6>8*p: n,p=n+1,2*p
    delta=h+n+4-4*p
    assert 0<=delta<=4*p-2
    a=6*p-2+delta//2
    if delta==0: return None,n+1,n,delta
    if delta%2==0: return a,n+2,n,delta
    nxt,k=drain(a)
    return nxt,n+2+k,n,delta

def source_first_return(h: int) -> tuple[int | None, int]:
    a,b=h,0
    count=0
    while True:
        nxt=source_next(a,b)
        if nxt is None: return None,count
        a,b=nxt
        count+=1
        if b==0: return a,count
